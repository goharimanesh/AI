دیتاست Ionosphere توسط سیستم راداری Goose Bay, Labrador جمع‌آوری شده و هدف آن تشخیص وجود ساختار الکترون آزاد در یونوسفر است.

  • 📡 پیکربندی سیستم: آرایهٔ فازی با ۱۶ آنتن فرکانس بالا، توان کل حدود ۶.۴ کیلووات
  • 🔊 پردازش سیگنال: سیگنال دریافتی با تابع خودهمبستگی پردازش می‌شود (زمان پالس + شمارهٔ پالس)
  • 📦 حجم نمونه: ۳۵۱ مشاهده، ۳۴ ویژگی پیوسته، بدون مقدار گمشده

پایتون (مخزن رسمی UCI):

import numpy as np
from ucimlrepo import fetch_ucirepo

# دریافت دیتاست Ionosphere از مخزن UCI (شناسه 52)
ion = fetch_ucirepo(id=52)
X = ion.data.features.values              # آرایه 351×34
y_raw = ion.data.targets.values.ravel()   # برچسب‌های 'g' یا 'b'
y = (y_raw == 'g').astype(int)            # تبدیل: g→1 و b→0
print(f"تعداد نمونه: {X.shape[0]}, تعداد ویژگی: {X.shape[1]}")
💡 اگر نمی‌خواهید ucimlrepo نصب کنید، فایل ionosphere.data را از UCI دانلود و با np.loadtxt بخوانید.

۴ تقسیم داده (حال در برابر آینده)

مطابق مثال رسمی MathWorks: ۳۰۰ نمونهٔ اول برای آموزش و ۵۱ نمونهٔ آخر برای شبیه‌سازی داده‌های آینده استفاده می‌شود.

% 300 نمونهٔ اول برای آموزش (حال)
prsntX = X(1:300,:);  prsntY = Y(1:300);
% 51 نمونهٔ آخر برای آزمون (آینده)
ftrX   = X(301:end,:); ftrY  = Y(301:end);
# 300 نمونهٔ اول برای آموزش
prsntX, prsntY = X[:300], y[:300]
# 51 نمونهٔ آخر برای آزمون
ftrX,   ftrY   = X[300:], y[300:]

۵ آموزش SVM (استانداردسازی الزامی است)

% آموزش مدل SVM با استانداردسازی خودکار ویژگی‌ها
svmMdl = fitcsvm(prsntX, prsntY, 'Standardize', true);
% نمایش ترتیب کلاس‌ها: {'b'; 'g'} → کلاس منفی b، کلاس مثبت g
disp(svmMdl.ClassNames);
from sklearn.svm import SVC
from sklearn.preprocessing import StandardScaler
from sklearn.pipeline import make_pipeline

# ساخت خط لوله: استانداردسازی + SVM با کرنل RBF
svm_clf = make_pipeline(
    StandardScaler(),                        # استانداردسازی ویژگی‌ها
    SVC(kernel='rbf', C=1.0, gamma='scale', probability=True)
)
svm_clf.fit(prsntX, prsntY)                 # آموزش مدل
⚠️ نکتهٔ کلیدی: ویژگی‌های Ionosphere مقیاس‌های بسیار متفاوتی دارند و SVM به مقیاس حساس است، پس استانداردسازی الزامی است. در متلب ترتیب ClassNames برابر {'b';'g'} است که با ستون‌های ۱ و ۲ خروجی score مطابقت دارد.

۶ پیش‌بینی و امتیازدهی

% پیش‌بینی برچسب و امتیاز برای داده‌های آینده
[labels, scores] = predict(svmMdl, ftrX);
% scores(:,1) → امتیاز کلاس b
% scores(:,2) → امتیاز کلاس g
pred_g = labels == 'g';
accuracy = mean(pred_g == (ftrY == 'g'));
fprintf('دقت روی داده‌های آینده: %.2f%%\n', accuracy*100);
# پیش‌بینی برچسب کلاس
pred_labels = svm_clf.predict(ftrX)
accuracy = (pred_labels == ftrY).mean()
print(f"دقت روی داده‌های آینده: {accuracy*100:.2f}%")

# امتیاز احتمال کلاس مثبت (g)
scores_g = svm_clf.predict_proba(ftrX)[:, 1]
📈 نمودار مقایسهٔ دقت تقریبی برای چند پیکربندی رایج را نشان می‌دهد (مقادیر نمونه؛ اجرای واقعی ممکن است تفاوت جزئی داشته باشد).

۷ قوی‌ترین مصورسازی: مرز تصمیم + بردارهای پشتیبان

با ۳۴ بُعد نمی‌توان مرز تصمیم را مستقیماً رسم کرد. روش استاندارد: کاهش ابعاد با PCA به ۲ بُعد.

import matplotlib.pyplot as plt
from sklearn.decomposition import PCA
from sklearn.inspection import DecisionBoundaryDisplay
from sklearn.svm import SVC

# کاهش ابعاد به 2 مؤلفهٔ اصلی
pca = PCA(n_components=2)
X_2d = pca.fit_transform(prsntX)

# آموزش SVM دوبعدی صرفاً برای مصورسازی
svm_2d = SVC(kernel='rbf', C=1.0, gamma='scale').fit(X_2d, prsntY)

fig, ax = plt.subplots(figsize=(10, 7))
# رسم مرز تصمیم و حاشیه‌ها
DecisionBoundaryDisplay.from_estimator(
    svm_2d, X_2d, response_method="predict",
    plot_method="contour", levels=[-1, 0, 1],
    colors="k", linestyles=["--", "-", "--"],
    alpha=0.6, ax=ax
)
# نقاط داده
ax.scatter(X_2d[:,0], X_2d[:,1], c=prsntY, cmap='coolwarm',
           edgecolors='k', s=40)
# بردارهای پشتیبان با دایره
ax.scatter(svm_2d.support_vectors_[:,0], svm_2d.support_vectors_[:,1],
           s=120, linewidth=1.5, facecolors='none',
           edgecolors='k', label='بردارهای پشتیبان')
ax.set_xlabel(f'PC1 ({pca.explained_variance_ratio_[0]*100:.1f}%)')
ax.set_ylabel(f'PC2 ({pca.explained_variance_ratio_[1]*100:.1f}%)')
ax.set_title('SVM با کرنل RBF روی Ionosphere — تصویر دوبعدی PCA')
ax.legend()
plt.tight_layout(); plt.show()
% کاهش ابعاد با PCA
[coeff, score2d] = pca(prsntX);
X2 = score2d(:,1:2);

% آموزش SVM دوبعدی برای مصورسازی
svmVis = fitcsvm(X2, prsntY, 'Standardize', true, 'KernelFunction', 'rbf');

% ساخت شبکهٔ نقاط برای رسم مرز
x1 = linspace(min(X2(:,1))-1, max(X2(:,1))+1, 200);
x2 = linspace(min(X2(:,2))-1, max(X2(:,2))+1, 200);
[X1g, X2g] = meshgrid(x1, x2);
[~, scoreGrid] = predict(svmVis, [X1g(:), X2g(:)]);
scoreGrid = reshape(scoreGrid(:,2), size(X1g));

% رسم نمودار
figure('Position',[100 100 900 650]);
contourf(X1g, X2g, scoreGrid, 20, 'LineColor', 'none'); hold on;
contour(X1g, X2g, scoreGrid, [0.5 0.5], 'k-', 'LineWidth', 2);
gscatter(X2(:,1), X2(:,2), prsntY, 'rb', 'o^', 8);
title('SVM با کرنل RBF روی Ionosphere — تصویر PCA');
xlabel('PC1'); ylabel('PC2'); hold off;

۸ تنظیم پارامترها و بهترین روش‌ها

پارامترنقشمحدودهٔ پیشنهادی
Cضریب جریمه؛ هرچه بیشتر، سخت‌گیرتر۰.۰۱ تا ۱۰۰
gammaعرض کرنل RBF؛ هرچه بیشتر، پیچیده‌ترscale، ۰.۰۰۱ تا ۱
kernelنوع تابع کرنلlinear / rbf / poly
from sklearn.model_selection import GridSearchCV

# شبکهٔ جستجو برای پارامترهای C و gamma
param_grid = {'svc__C': [0.1, 1, 10],
              'svc__gamma': ['scale', 0.01, 0.1]}
gs = GridSearchCV(svm_clf, param_grid, cv=5, scoring='accuracy')
gs.fit(prsntX, prsntY)
print(gs.best_params_, gs.best_score_)
