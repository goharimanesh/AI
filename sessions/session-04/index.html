🧬 Crossover و Mutation

🔀 Crossover

دو والد ترکیب می‌شوند تا فرزندانی با ویژگی‌های هر دو تولید کنند.

$$\text{Child} = \alpha \cdot P_1 + (1-\alpha) \cdot P_2$$ ترکیب حسابی دو والد

⚡ Mutation

تغییر تصادفی در یک ژن برای حفظ تنوع و جلوگیری از همگرایی زودهنگام.

$$x' = x + \mathcal{N}(0, \sigma^2)$$ جهش با نویز گاوسی

function GA():
    population = initialize_random(N, bounds)
    for gen in range(max_gen):
        fitness = [evaluate(ind) for ind in population]
        parents = roulette_select(population, fitness)
        offspring = []
        for i in range(0, N, 2):
            child1, child2 = crossover(parents[i], parents[i+1])
            mutate(child1, mutation_rate)
            mutate(child2, mutation_rate)
            offspring.append(child1, child2)
        population = offspring
    return best(population, fitness)
# توضیح گام‌به‌گام

1. جمعیت اولیه: N کروموزوم تصادفی در بازه [min, max]
2. ارزیابی: محاسبه Fitness هر کروموزوم
3. انتخاب: انتخاب N والد با Roulette یا Tournament
4. Crossover: ترکیب جفت‌های والد → فرزندان
5. Mutation: تغییر تصادفی برخی ژن‌ها
6. جایگزینی: جمعیت جدید جای قدیمی
7. تکرار گام‌های ۲-۶ تا شرط توقف
8. خروجی: بهترین کروموزوم

import numpy as np

def fitness(chrom, x_true, t):
    c, k = chrom
    x_model = np.exp(-c*t) * np.cos(k*t)
    mse = np.mean((x_true - x_model)**2)
    return 1 / (1 + mse)

def ga(x_true, t, N=30, gens=50, pc=0.8, pm=0.1):
    bounds = np.array([[0, 5], [0.5, 10]])
    pop = np.random.uniform(bounds[:,0], bounds[:,1], (N, 2))
    best_fit = 0; best_ind = None
    for gen in range(gens):
        fits = np.array([fitness(ind, x_true, t) for ind in pop])
        idx = np.argmax(fits)
        if fits[idx] > best_fit:
            best_fit = fits[idx]; best_ind = pop[idx].copy()
        probs = fits / fits.sum()
        parents_idx = np.random.choice(N, N, p=probs)
        new_pop = []
        for i in range(0, N, 2):
            p1, p2 = pop[parents_idx[i]], pop[parents_idx[i+1]]
            alpha = np.random.rand()
            c1 = alpha*p1 + (1-alpha)*p2
            c2 = (1-alpha)*p1 + alpha*p2
            if np.random.rand() < pm: c1 += np.random.randn(2)*0.1
            if np.random.rand() < pm: c2 += np.random.randn(2)*0.1
            new_pop.extend([c1, c2])
        pop = np.clip(new_pop, bounds[:,0], bounds[:,1])
    return best_ind, best_fit

c_true, k_true = 0.8, 3.0
t = np.linspace(0, 8, 300)
x_true = np.exp(-c_true*t) * np.cos(k_true*t)
best, fit = ga(x_true, t)
print(f"c_est = {best[0]:.3f}, k_est = {best[1]:.3f}")
